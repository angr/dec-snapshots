unsigned char * cgc_b2hex(unsigned char a0, unsigned char *a1)
{
    if (((char)(a0) >> 4) < 10)
        *a1 = ((char)(a0) >> 4) + 48;
    else
        *a1 = ((char)(a0) >> 4) + 55;
    if ((a0 & 15) < 10)
        a1[1] = (a0 & 15) + 48;
    else
        a1[1] = (a0 & 15) + 55;
    a1[2] = 0;
    return a1;
}
