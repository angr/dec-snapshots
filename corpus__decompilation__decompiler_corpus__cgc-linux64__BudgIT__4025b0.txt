long long cgc_strcpy(unsigned char *a0, unsigned char *a1)
{
    unsigned char *iter;  // [bp-0x20]
    unsigned char *v1;  // [bp-0x20]
    unsigned char *i;  // [bp-0x18]

    i = a1;
    for (iter = a0; *i; i += 1)
    {
        *iter = *i;
        iter += 1;
    }
    v1 = iter + 1;
    *iter = 0;
    return v1 - a0 - 1;
}
